cmake_minimum_required(VERSION 3.21)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" IMAGEBATCH_PROJECT_VERSION LIMIT_COUNT 1)
project(ImageBatch VERSION ${IMAGEBATCH_PROJECT_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
include(CTest)
option(IMAGEBATCH_BUILD_APP "Build the Qt desktop application" ON)
add_library(imagebatch_runtime src/threadpool.cpp src/resourcebudget.cpp)
target_include_directories(imagebatch_runtime PUBLIC src)
target_link_libraries(imagebatch_runtime PUBLIC Threads::Threads)
target_compile_definitions(imagebatch_runtime PUBLIC IMAGEBATCH_VERSION="${PROJECT_VERSION}")
if(BUILD_TESTING)
  add_executable(runtime_tests tests/test_runtime.cpp)
  target_link_libraries(runtime_tests PRIVATE imagebatch_runtime)
  add_test(NAME runtime_tests COMMAND runtime_tests)
  set_tests_properties(runtime_tests PROPERTIES TIMEOUT 30)
endif()
if(NOT IMAGEBATCH_BUILD_APP)
  return()
endif()
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Concurrent)
add_library(imagebatch_core src/processor.cpp src/batchengine.cpp src/batchworker.cpp)
target_include_directories(imagebatch_core PUBLIC src)
target_link_libraries(imagebatch_core PUBLIC imagebatch_runtime Qt6::Core Qt6::Gui)
add_library(imagebatch_ui src/mainwindow.cpp src/preferences.cpp)
target_link_libraries(imagebatch_ui PUBLIC imagebatch_core Qt6::Widgets Qt6::Concurrent)
add_executable(ImageBatch WIN32 src/main.cpp src/packagecheck.cpp)
target_link_libraries(ImageBatch PRIVATE imagebatch_ui)
add_executable(ImageBatchBenchmark tools/benchmark.cpp)
target_link_libraries(ImageBatchBenchmark PRIVATE imagebatch_core)
if(WIN32)
  target_link_libraries(ImageBatchBenchmark PRIVATE psapi)
endif()
if(MSVC)
  target_compile_options(imagebatch_runtime PRIVATE /W4 /utf-8)
  target_compile_options(imagebatch_core PRIVATE /W4 /utf-8)
  target_compile_options(imagebatch_ui PRIVATE /W4 /utf-8)
else()
  target_compile_options(imagebatch_runtime PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(imagebatch_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(imagebatch_ui PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
  add_executable(imagebatch_tests tests/test_imagebatch.cpp)
  target_link_libraries(imagebatch_tests PRIVATE imagebatch_ui Qt6::Test)
  add_test(NAME imagebatch_tests COMMAND imagebatch_tests)
  set_tests_properties(imagebatch_tests PROPERTIES TIMEOUT 180)
endif()
install(TARGETS ImageBatch RUNTIME DESTINATION .)
